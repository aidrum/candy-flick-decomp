cmake_minimum_required(VERSION 3.22)
project(CandyFlickTracking LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(ANDROID)
    set(OpenCV_STATIC ON)
endif()
find_package(OpenCV 4.12 REQUIRED COMPONENTS core imgproc features2d calib3d video)
add_library(CandyFlickTracking SHARED tracker.cpp)
target_include_directories(CandyFlickTracking PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(CandyFlickTracking PRIVATE ${OpenCV_LIBS})
if(ANDROID)
    target_link_options(CandyFlickTracking PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384" "-Wl,--exclude-libs,ALL")
    target_compile_options(CandyFlickTracking PRIVATE -fvisibility=hidden)
endif()
